cmake_minimum_required(VERSION 3.22)
project(SonShellRemote VERSION 0.6.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
include(CTest)
set(SONY_SDK_DIR "$ENV{SONY_SDK_DIR}" CACHE PATH "Sony SDK package root containing app/ and external/")
option(SONSHELL_BUILD_BACKEND "Build SonShell backend (requires Sony SDK)" OFF)
option(SONSHELL_HEADLESS "Omit standalone OpenCV viewer; GUI Live View remains available" ON)
add_subdirectory(gui)
if(SONSHELL_BUILD_BACKEND)
  include(cmake/SonShellBackend.cmake)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
install(FILES packaging/sonshell-remote.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/sonshell-remote.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/sonshell-remote)
include(cmake/Packaging.cmake)
